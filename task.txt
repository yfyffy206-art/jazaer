<script>
const modal = document.getElementById("formModal");
const closeButton = document.getElementById("closeModal");
const form = document.getElementById("demoForm");
const warning = document.getElementById("formWarning");


// فتح النافذة
function openModal() {
  modal.classList.add("show");
  warning.classList.remove("show");


  document.body.style.overflow = "hidden";


  setTimeout(() => {
    document.getElementById("fullName").focus();
  }, 100);
}


// إغلاق النافذة
function closeModal() {
  modal.classList.remove("show");
  document.body.style.overflow = "";
}


// جميع أزرار التسجيل تفتح النموذج
document.querySelectorAll(".open-form").forEach(button => {
  button.addEventListener("click", openModal);
});


// زر X
closeButton.addEventListener("click", closeModal);


// إغلاق عند الضغط خارج النموذج
modal.addEventListener("click", event => {
  if (event.target === modal) {
    closeModal();
  }
});


/new 
 إغلاق بزر ESC
document.addEventListener("keydown", event => {
  if (
    event.key === "Escape" &&
    modal.classList.contains("show")
  ) {
    closeModal();
  }
});


// التحقق من الحقول
form.addEventListener("submit", event => {
  event.preventDefault();


  const name = document.getElementById("fullName").value.trim();
  const phone = document.getElementById("phone").value.trim();


  const choice = document.querySelector(
    'input[name="hasAccount"]:checked'
  );


  if (!name || !phone || !choice) {
    warning.textContent =
      "يرجى إدخال الاسم ورقم الهاتف واختيار إجابة.";


    warning.classList.add("show");
    return;
  }


  warning.classList.remove("show");


  alert(
    "تم اختبار النموذج بنجاح. هذه نسخة تجريبية ولم يتم إرسال أو حفظ أي بيانات."
  );


  form.reset();
  closeModal();
});
</script>
